        function calculateMaterials() {
            const length = parseFloat(document.getElementById('length').value);
            const width = parseFloat(document.getElementById('width').value);
            const height = parseFloat(document.getElementById('height').value);
            const waste = parseFloat(document.getElementById('waste').value) / 100 || 0.1;

            // Basement wall perimeter and area
            const perimeter = 2 * (length + width);
            const wallArea = perimeter * height; // sq ft for walls

            // Studs: 16" OC = every 1.333 ft, plus 4 corners, times waste
            const linearFtWalls = perimeter * height;
            const studs = Math.ceil((linearFtWalls / 1.333) + 4) * (1 + waste / 2); // /2 for waste on linear

            // Drywall: 4x8 sheets = 32 sq ft each, 10% waste
            const drywallSheets = Math.ceil(wallArea / 32 * (1 + waste));

            // Insulation: Assume R-13 batts ~23" wide x 8ft = ~12 sq ft per batt, for walls
            const insulationBatts = Math.ceil(wallArea / 12 * (1 + waste / 2));

            // Flooring: Plywood/OSB 4x8 = 32 sq ft, room area + waste
            const floorArea = length * width;
            const flooringSheets = Math.ceil(floorArea / 32 * (1 + waste));

            // Paint: 350 sq ft/gallon per coat, 2 coats on walls
            const paintGallons = Math.ceil(wallArea * 2 / 350 * (1 + waste / 4));

            // Populate table
            const tbody = document.querySelector('#materialTable tbody');
            tbody.innerHTML = `
                <tr><td>2x4 Studs (8ft)</td><td>${Math.round(studs)}</td><td>${(studs * 4.50).toFixed(0)} est. $ (at $4.50 ea)</td></tr>
                <tr><td>4x8 Drywall Sheets (1/2")</td><td>${drywallSheets}</td><td>${(drywallSheets * 13).toFixed(0)} est. $ (at $13 ea)</td></tr>
                <tr><td>R-13 Insulation Batts</td><td>${Math.round(insulationBatts)}</td><td>${(insulationBatts * 1.50).toFixed(0)} est. $ (at $1.50 ea)</td></tr>
                <tr><td>4x8 Plywood/OSB (Subfloor)</td><td>${flooringSheets}</td><td>${(flooringSheets * 25).toFixed(0)} est. $ (at $25 ea)</td></tr>
                <tr><td>Paint (Interior Latex)</td><td>${paintGallons} gallons</td><td>${(paintGallons * 30).toFixed(0)} est. $ (at $30/gal)</td></tr>
                <tr><td><strong>Total Est. Cost</strong></td><td>-</td><td><strong>$${(studs*4.50 + drywallSheets*13 + insulationBatts*1.50 + flooringSheets*25 + paintGallons*30).toFixed(0)}</strong></td></tr>
            `;

            document.getElementById('results').style.display = 'block';
            document.getElementById('results').scrollIntoView({ behavior: 'smooth' });
        }

        function exportPDF() {

        function shareResults() {
            if (navigator.share) {
                navigator.share({ title: 'My Basement Reno Calc', text: 'Check out this material list!', url: window.location.href });
            } else {
                prompt('Copy this link to share:', window.location.href);
            }
        }
