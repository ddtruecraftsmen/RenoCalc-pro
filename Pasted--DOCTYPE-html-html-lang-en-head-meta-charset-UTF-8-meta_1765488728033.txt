    <script>
        console.log('RenoCalc V1.0.0 loaded—form ready!'); // Debug: Check console for this

        function calculateMaterials() {
            console.log('Calculate button clicked!'); // Debug log
            const length = parseFloat(document.getElementById('length').value);
            const width = parseFloat(document.getElementById('width').value);
            const height = parseFloat(document.getElementById('height').value);
            const waste = parseFloat(document.getElementById('waste').value) / 100 || 0.1;

            if (!length || !width || !height) {
                alert('Hey—fill in length, width, and height to get started!');
                return;
            }

            const loading = document.getElementById('loading');
            const table = document.getElementById('materialTable');
            const proTip = document.getElementById('proTip');
            const exportBtn = document.getElementById('exportBtn');
            const shareBtn = document.getElementById('shareBtn');
            const results = document.getElementById('results');

            loading.style.display = 'block';
            table.style.display = 'none';
            proTip.style.display = 'none';
            exportBtn.style.display = 'none';
            shareBtn.style.display = 'none';

            setTimeout(() => { // Quick "thinking" animation
                const perimeter = 2 * (length + width);
                const wallArea = perimeter * height;
                const floorArea = length * width;

                // Studs per your logic: 16" OC + corners
                const linearWallFt = perimeter * height;
                const studs = Math.ceil((linearWallFt / 1.333) + 4) * (1 + waste / 2);

                // Drywall + waste
                const drywallSheets = Math.ceil(wallArea / 32 * (1 + waste));

                // Flooring + waste
                const flooringSheets = Math.ceil(floorArea / 32 * (1 + waste));

                // Paint: 2 coats
                const paintGallons = Math.ceil(wallArea * 2 / 350 * (1 + waste / 4));

                // Costs (tweak as needed)
                const studCost = 4.50; // per 8ft stud
                const drywallCost = 13; // per sheet
                const flooringCost = 25; // per sheet
                const paintCost = 30; // per gallon

                const totalCost = (studs * studCost) + (drywallSheets * drywallCost) + (flooringSheets * flooringCost) + (paintGallons * paintCost);

                const tbody = document.querySelector('#materialTable tbody');
                tbody.innerHTML = `
                    <tr><td>2x4 Studs (8ft)</td><td>${Math.round(studs)}</td><td>$${Math.round(studs * studCost)}</td></tr>
                    <tr><td>4x8 Drywall Sheets (1/2")</td><td>${drywallSheets}</td><td>$${Math.round(drywallSheets * drywallCost)}</td></tr>
                    <tr><td>4x8 Plywood/OSB (Flooring)</td><td>${flooringSheets}</td><td>$${Math.round(flooringSheets * flooringCost)}</td></tr>
                    <tr><td>Interior Paint (2 coats)</td><td>${paintGallons} gallons</td><td>$${Math.round(paintGallons * paintCost)}</td></tr>
                    <tr><td><strong>Grand Total</strong></td><td>-</td><td><strong>$${Math.round(totalCost)}</strong></td></tr>
                `;

                loading.style.display = 'none';
                table.style.display = 'table';
                proTip.style.display = 'block';
                exportBtn.style.display = 'inline-block';
                shareBtn.style.display = 'inline-block';
                results.style.display = 'block';
                results.scrollIntoView({ behavior: 'smooth' });

                console.log(`Calc complete: ${Math.round(studs)} studs, $${Math.round(totalCost)} total`); // Debug
            }, 800);
        }

        function exportPDF() {
            window.print();
        }

        function shareResults() {
            const shareText = `Basement reno calc: ${Math.round(studs)} studs + more for a ${length} x ${width} room = ~$${Math.round(totalCost)} total. Try it!`;
            if (navigator.share) {
                navigator.share({ title: 'RenoCalc Pro', text: shareText, url: window.location.href });
            } else {
                navigator.clipboard.writeText(shareText + ' ' + window.location.href).then(() => alert('Copied—share away!'));
            }
        }

        // Ensure everything loads
        document.addEventListener('DOMContentLoaded', () => {
            console.log('DOM loaded—inputs visible?');
            document.getElementById('length').focus();
        });
