        function calculateMaterials() {
            const length = parseFloat(document.getElementById('length').value);
            const width = parseFloat(document.getElementById('width').value);
            const height = parseFloat(document.getElementById('height').value);
            const waste = parseFloat(document.getElementById('waste').value) / 100 || 0.1;

            if (!length || !width || !height) {
                alert('Please enter length, width, and height!');
                return;
            }

            document.getElementById('loading').style.display = 'block';
            document.getElementById('materialTable').style.display = 'none';
            document.getElementById('proTip').style.display = 'none';

            // Simulate quick calc delay for polish
            setTimeout(() => {
                const perimeter = 2 * (length + width);
                const wallArea = perimeter * height;
                const floorArea = length * width;

                // Studs: 16" OC (1.333 ft) + 4 corners, waste on count
                const linearWallFt = perimeter * height;
                const studs = Math.ceil((linearWallFt / 1.333) + 4) * (1 + waste / 2);

                // Drywall: 32 sq ft/sheet + waste
                const drywallSheets = Math.ceil(wallArea / 32 * (1 + waste));

                // Flooring: 32 sq ft/sheet + waste
                const flooringSheets = Math.ceil(floorArea / 32 * (1 + waste));

                // Paint: 350 sq ft/gal per coat, 2 coats + minor waste
                const paintGallons = Math.ceil(wallArea * 2 / 350 * (1 + waste / 4));

                // Costs (easy to tweak)
                const studCost = 4.50;
                const drywallCost = 13;
                const flooringCost = 25;
                const paintCost = 30;

                const tbody = document.querySelector('#materialTable tbody');
                tbody.innerHTML = `
                    <tr><td>2x4 Studs (8ft)</td><td>${Math.round(studs)}</td><td>$${Math.round(studs * studCost)}</td></tr>
                    <tr><td>4x8 Drywall Sheets (1/2")</td><td>${drywallSheets}</td><td>$${Math.round(drywallSheets * drywallCost)}</td></tr>
                    <tr><td>4x8 Plywood/OSB (Flooring)</td><td>${flooringSheets}</td><td>$${Math.round(flooringSheets * flooringCost)}</td></tr>
                    <tr><td>Interior Paint (2 coats)</td><td>${paintGallons} gallons</td><td>$${Math.round(paintGallons * paintCost)}</td></tr>
                    <tr><td><strong>Grand Total</strong></td><td>-</td><td><strong>$${(studs*studCost + drywallSheets*drywallCost + flooringSheets*flooringCost + paintGallons*paintCost).toFixed(0)}</strong></td></tr>
                `;

                document.getElementById('loading').style.display = 'none';
                document.getElementById('materialTable').style.display = 'table';
                document.getElementById('proTip').style.display = 'block';
                document.getElementById('exportBtn').style.display = 'inline-block';
                document.getElementById('shareBtn').style.display = 'inline-block';
                document.getElementById('results').style.display = 'block';
                document.getElementById('results').scrollIntoView({ behavior: 'smooth' });
            }, 500);
        }

        function exportPDF() {

        }
